SECTIONS {
	. = 0x02000000;
    .text :
    ALIGN(4)
    {
		asm/crt0.o(.text);
    } =0

    .rodata :
    ALIGN(4)
    {
		data/payload.o(.rodata);
    } =0

    . = 0x3000000;
    iwram (NOLOAD) :
    ALIGN(4)
    {
		arm_LZ77UnCompWRAM_Buffer = .;
		. += 0x130;
		arm_LZ77UnCompWRAM_FuncStart = .;
		. += 0x94;
		arm_LZ77UnCompWRAM_BufferEnd = .;
    }
    /* Standalone debug cartridge assets; file offset 0x40000. */
    .debug_graphics 0x08040000 : AT(0x02040000)
    {
        data/debug_graphics.o(.debug_graphics);
    }
    ASSERT(SIZEOF(.debug_graphics) == 0 || SIZEOF(.debug_graphics) == 0x18000,
           "debug graphics banks have the wrong size")

    /DISCARD/ :
    {
        *(*);
    }
}
